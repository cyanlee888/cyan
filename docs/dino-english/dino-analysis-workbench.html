Push 归因口径与数据缺口 Definitions & Gaps

现有 Firebase 自动事件已经能回答“权限、收到、打开、划掉”,自定义 label 能拆场景与轮次;但业务事件没有携带 push_id / task_id,所以“回访与任务完成”目前只能做同设备的时序归因。

补齐完整归因的最小改造:发送侧生成 push_id,并同时写入 scene、round、send_batch、target_task、deeplink;客户端在 receive / open 及落地页首个业务事件透传同一 push_id;另保留 5%~10% 同条件不发送组,才能计算 Push 对 D1 / D3、学习与付费的真实增量。

SELECT
  PARSE_DATE('%Y%m%d', REGEXP_EXTRACT(_TABLE_SUFFIX,r'(\d{8})$')) AS d,
  device.operating_system AS os,
  COUNTIF(event_name='notification_receive') AS receive,
  COUNTIF(event_name='notification_open') AS open,
  COUNTIF(event_name='notification_dismiss') AS dismiss,
  COUNTIF(event_name='notification_foreground') AS foreground,
  ROUND(SAFE_DIVIDE(COUNTIF(event_name='notification_open'),
                    COUNTIF(event_name='notification_receive'))*100,2) AS open_rate
FROM `dino-english-497507.analytics_538991439.events_*`
WHERE REGEXP_EXTRACT(_TABLE_SUFFIX,r'(\d{8})$') BETWEEN '20260716' AND '20260726'
  AND event_name IN ('notification_receive','notification_open',
                     'notification_dismiss','notification_foreground')
GROUP BY 1,2 ORDER BY 1,2

WITH e AS (
  SELECT event_name, user_pseudo_id,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key='label') AS label
  FROM `dino-english-497507.analytics_538991439.events_*`
  WHERE REGEXP_EXTRACT(_TABLE_SUFFIX,r'(\d{8})$') BETWEEN '20260716' AND '20260726'

    AND event_name IN ('notification_receive','notification_open')
)
SELECT label,
  COUNTIF(event_name='notification_receive') receive,
  COUNTIF(event_name='notification_open') open,
  ROUND(SAFE_DIVIDE(COUNTIF(event_name='notification_open'),
                    COUNTIF(event_name='notification_receive'))*100,2) open_rate
FROM e GROUP BY 1 ORDER BY receive DESC
③ 打开后 24h 学习 / 完课(时序归因代理) 
WITH opens AS (
  SELECT user_pseudo_id, event_timestamp open_ts,
    (SELECT value.string_value FROM UNNEST(event_params) WHERE key='label') label
  FROM `dino-english-497507.analytics_538991439.events_*`
  WHERE REGEXP_EXTRACT(_TABLE_SUFFIX,r'(\d{8})$') BETWEEN '20260716' AND '20260725'
    AND device.operating_system='Android'
    AND event_name='notification_open'
),
biz AS (
  SELECT user_pseudo_id, event_timestamp, event_name
  FROM `dino-english-497507.analytics_538991439.events_*`
  WHERE REGEXP_EXTRACT(_TABLE_SUFFIX,r'(\d{8})$') BETWEEN '20260716' AND '20260726'
)
SELECT o.label, COUNT(*) opens,
  COUNTIF(EXISTS(
    SELECT 1 FROM biz b
    WHERE b.user_pseudo_id=o.user_pseudo_id
      AND b.event_timestamp BETWEEN o.open_ts AND o.open_ts+24*60*60*1000000
      AND b.event_name IN ('class_lesson_start','dino_assistant_progress')
  )) learning_24h,
  COUNTIF(EXISTS(
    SELECT 1 FROM biz b
    WHERE b.user_pseudo_id=o.user_pseudo_id
      AND b.event_timestamp BETWEEN o.open_ts AND o.open_ts+24*60*60*1000000
      AND b.event_name IN ('class_lesson_end','trial_lesson_complete')
  )) lesson_complete_24h
FROM opens o GROUP BY 1 ORDER BY opens DESC
读数边界:① 当前有 11 个发送日、Android 148 次打开，分轮次结论仍需扩样；② notification_receive 是 Firebase 客户端自动事件，不等于服务端“发送成功”；③ 同一设备在 24 小时内自然回访也会被计入，没有不发送对照组前不可宣称增量；④ iOS 只有 16 次 open，本页不展示虚假的点击率。SQL 用日期正则同时覆盖日表与 07-26 intraday；24h 归因只取打开至 07-25。
口径说明与可复跑 SQL Definitions & Queries

所有查询在 BigQuery 项目 dino-english-497507 跑,GA4 数据集 analytics_538991439(日表 events_YYYYMMDD + 当日 events_intraday_*),数据侧宽表在 de_dwd。跑法:bq query --use_legacy_sql=false '…'。

口径速查:① GA4 注册率 = 首启当日 signup_result 设备 ÷ 当日 first_open 设备(user_pseudo_id 去重,cohort 化,国家取首启事件 geo.country;intraday 表 geo 可能缺失,分国家时改用 install_country 参数);② 宽表注册率 = is_registered ÷ is_installed(de_dwd.product_conversion_funnel_user,AppsFlyer 安装锚点,身份桥 appsflyer_id→user_pseudo_id→uid);③ 激活 = 安装能关联到 first_open 且日期 ≥ 安装日;④ 留存 = cohort 首启日分组,第 N 天有任意事件。

WITH ev AS (
  SELECT event_name, user_pseudo_id, geo.country AS country,
    PARSE_DATE('%Y%m%d',REGEXP_EXTRACT(_TABLE_SUFFIX,r'(\d{8})$')) d
  FROM `dino-english-497507.analytics_538991439.events_*`
  WHERE REGEXP_EXTRACT(_TABLE_SUFFIX,r'(\d{8})$') BETWEEN '20260710' AND '20260726'
    AND event_name IN ('first_open','signup_result')
),
firsts AS (
  SELECT user_pseudo_id, MIN(d) cd,
    ARRAY_AGG(IF(event_name='first_open',country,NULL) IGNORE NULLS ORDER BY d LIMIT 1)[SAFE_OFFSET(0)] country
  FROM ev WHERE event_name='first_open' GROUP BY 1
),
su AS (SELECT user_pseudo_id, MIN(d) sd FROM ev WHERE event_name='signup_result' GROUP BY 1)
SELECT f.cd, f.country, COUNT(*) installs,
  COUNTIF(s.sd=f.cd) su_same_day,
  ROUND(COUNTIF(s.sd=f.cd)/COUNT(*)*100,1) rate
FROM firsts f LEFT JOIN su s USING(user_pseudo_id)
GROUP BY 1,2 ORDER BY 1,2

⑥ 留存(cohort × 第 N 天) 
WITH ev AS ( ... 全事件,日表+intraday ... ),
act AS (SELECT DISTINCT user_pseudo_id, d FROM ev),
firsts AS (SELECT user_pseudo_id, MIN(d) cd FROM ev WHERE event_name='first_open' GROUP BY 1)
SELECT f.cd, COUNT(DISTINCT f.user_pseudo_id) n,
  COUNT(DISTINCT IF(a.d=DATE_ADD(f.cd,INTERVAL 1 DAY),a.user_pseudo_id,NULL)) d1,
  COUNT(DISTINCT IF(a.d=DATE_ADD(f.cd,INTERVAL 3 DAY),a.user_pseudo_id,NULL)) d3,
  COUNT(DISTINCT IF(a.d=DATE_ADD(f.cd,INTERVAL 7 DAY),a.user_pseudo_id,NULL)) d7
FROM firsts f LEFT JOIN act a USING(user_pseudo_id) GROUP BY 1 ORDER BY 1

① 核心免费转化漏斗(报告 → 计划 → 订阅) 
WITH e AS (
  SELECT event_name, user_pseudo_id,
    (SELECT x.value.string_value FROM UNNEST(event_params) x WHERE x.key='view_source') view_source,
    (SELECT x.value.string_value FROM UNNEST(event_params) x WHERE x.key='lesson_type') lesson_type,
    (SELECT x.value.string_value FROM UNNEST(event_params) x WHERE x.key='button_id') button_id,
    (SELECT x.value.string_value FROM UNNEST(event_params) x WHERE x.key='plan_source') plan_source,
    (SELECT x.value.string_value FROM UNNEST(event_params) x WHERE x.key='subscription_source') sub_source,
    (SELECT x.value.string_value FROM UNNEST(event_params) x WHERE x.key='purchase_result') purchase_result,
    (SELECT x.value.string_value FROM UNNEST(event_params) x WHERE x.key='purchase_type') purchase_type
  FROM `dino-english-497507.analytics_538991439.events_*`
  WHERE REGEXP_EXTRACT(_TABLE_SUFFIX,r'(\d{8})$') BETWEEN '20260716' AND '20260726'
)
SELECT
  COUNT(DISTINCT IF(event_name='report_page_view' AND lesson_type='trial' AND view_source='course_flow', user_pseudo_id, NULL)) s1_report,
  COUNT(DISTINCT IF(event_name='button_click' AND button_id='trial_report_get_plan', user_pseudo_id, NULL)) s2_get_plan,
  COUNT(DISTINCT IF(event_name='study_plan_page_view' AND plan_source='trial_report', user_pseudo_id, NULL)) s3_plan,
  COUNT(DISTINCT IF(event_name='subscription_page_view' AND sub_source='study_plan', user_pseudo_id, NULL)) s4_paywall,
  COUNT(DISTINCT IF(event_name='subscription_checkout_start' AND sub_source='study_plan', user_pseudo_id, NULL)) s5_checkout,
  COUNT(DISTINCT IF(event_name='subscription_purchase_result' AND purchase_result='success' AND purchase_type='initial_purchase', user_pseudo_id, NULL)) s6_purchase
FROM e
② 挽留 / 每日折扣(按 subscription_source) 
SELECT
  (SELECT x.value.string_value FROM UNNEST(event_params) x WHERE x.key='subscription_source') source,
  COUNTIF(event_name='discount_offer_view') offer_view,
  COUNTIF(event_name='subscription_checkout_start') checkout_start,
  COUNTIF(event_name='subscription_purchase_result'
    AND (SELECT x.value.string_value FROM UNNEST(event_params) x WHERE x.key='purchase_result')='success') success
FROM `dino-english-497507.analytics_538991439.events_*`
WHERE REGEXP_EXTRACT(_TABLE_SUFFIX,r'(\d{8})$') BETWEEN '20260716' AND '20260726'
  AND event_name IN ('discount_offer_view','subscription_checkout_start','subscription_purchase_result')
GROUP BY 1
HAVING source IN ('winback_payment_cancel','winback_paywall_close','daily_offer')
③ Paywall 年+月结构(周期 × 客单价) 
SELECT
  (SELECT x.value.string_value FROM UNNEST(event_params) x WHERE x.key='billing_cycle') cycle,
  COUNTIF(event_name='subscription_checkout_start') checkout_start,
  COUNTIF(event_name='subscription_purchase_result'
    AND (SELECT x.value.string_value FROM UNNEST(event_params) x WHERE x.key='purchase_result')='success'
    AND (SELECT x.value.string_value FROM UNNEST(event_params) x WHERE x.key='purchase_type')='initial_purchase') first_purchase,
  ROUND(AVG(IF(event_name='subscription_purchase_result',
    (SELECT x.value.double_value FROM UNNEST(event_params) x WHERE x.key='price'), NULL)),2) avg_price
FROM `dino-english-497507.analytics_538991439.events_*`
WHERE REGEXP_EXTRACT(_TABLE_SUFFIX,r'(\d{8})$') BETWEEN '20260716' AND '20260726'
  AND event_name IN ('subscription_checkout_start','subscription_purchase_result')
GROUP BY 1
