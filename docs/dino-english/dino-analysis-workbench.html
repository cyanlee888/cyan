① 金币余额分布 
SELECT CASE
  WHEN balance = 0 THEN '0'
  WHEN balance BETWEEN 1 AND 49 THEN '1-49'
  WHEN balance BETWEEN 50 AND 99 THEN '50-99'
  WHEN balance BETWEEN 100 AND 299 THEN '100-299'
  WHEN balance BETWEEN 300 AND 999 THEN '300-999'
  ELSE '1000+'
END AS bucket,
COUNT(*) users
FROM dino_coin_account
GROUP BY bucket;
② 手办门槛 → 抽取 
SELECT
  COUNT(*) AS progress_users,
  SUM(total_correct >= 5) AS reached_threshold,
  COUNT(DISTINCT d.user_id) AS draw_users,
  SUM(p.pending_draws) AS pending_draws
FROM play_capsule_progress p
LEFT JOIN (
  SELECT DISTINCT user_id
  FROM play_capsule_draw_record
) d USING(user_id);
③ 皮肤购买与当前穿戴 
WITH buyers AS (
  SELECT DISTINCT ui.user_id
  FROM user_item ui
  JOIN item_catalog ic ON ic.id=ui.item_id
  WHERE ic.item_type='SKIN'
    AND ui.source='SHOP_PURCHASE'
),
equipped_purchased AS (
  SELECT DISTINCT ue.user_id
  FROM user_equip ue
  JOIN user_item ui
    ON ui.user_id=ue.user_id AND ui.item_id=ue.item_id
  WHERE ui.source='SHOP_PURCHASE'
)
SELECT
  (SELECT COUNT(*) FROM buyers) buyers,
  (SELECT COUNT(*) FROM equipped_purchased) using_purchase;
必须补的数据能力:① 新增换装事件流水，避免用 user_equip.updated_at 猜测行为；② 金币账户初始化必须记 opening_balance 流水；③ 皮肤订单保留成交时 price，而不是事后回看 item_catalog 当前价；④ 手办重复抽取需要记录补偿结果，才能衡量重复体验是否伤害第 2 抽。
